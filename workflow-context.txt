def read_html_file(filename: str) -> str:
    file_path = os.path.join("frontend", "templates", filename)

    try:
        with open(file_path, "r", encoding="utf-8") as f:
            return f.read()
    except FileNotFoundError:
        logger.error(f"فایل {filename} پیدا نشد")
        return f"""
        <!DOCTYPE html>
        <html lang="fa" dir="rtl">
        <head>
            <meta charset="UTF-8">
            <title>خطا</title>
            <link href="https://cdn.jsdelivr.net/npm/bootstrap@5.3.0/dist/css/bootstrap.rtl.min.css" rel="stylesheet">
        </head>

            <div class="container">
                <div class="alert alert-danger">
                    <h1>⚠️ فایل {filename} پیدا نشد!</h1>
                    <p>لطفاً مسیر فایل‌های frontend/templates را بررسی کنید.</p>

# ============================================
# تنظیمات اتصال به دیتابیس PostgreSQL

engine = create_engine(
    settings.database_url,
    echo=settings.debug,
    pool_size=10,
    max_overflow=20,
    pool_pre_ping=True,
    pool_recycle=3600,
)

SessionLocal = sessionmaker(
    autocommit=False,
    autoflush=False,
    bind=engine
)

# ============================================
# تابع دریافت سشن دیتابیس برای Dependency
# ============================================
def get_db() -> Generator[Session, None, None]:
    db = SessionLocal()
    try:
        yield db
    except Exception as e:
        logger.error(f"خطا در سشن دیتابیس: {e}")
        db.rollback()
        raise
    finally:
        db.close()

from __future__ import annotations

from datetime import datetime
from typing import Optional

from sqlalchemy import Boolean, DateTime, ForeignKey, Integer, String, Text, func
from sqlalchemy.orm import Mapped, mapped_column, relationship

from app.models.base import Base



class WorkflowTransition(Base):
    __tablename__ = "workflow_transitions"

    id: Mapped[int] = mapped_column(Integer, primary_key=True, index=True)
    repair_order_id: Mapped[int] = mapped_column(
        ForeignKey("repair_orders.id", ondelete="CASCADE"),
        nullable=False,
        index=True,
    )

    from_user_id: Mapped[int] = mapped_column(
        ForeignKey("users.id", ondelete="RESTRICT"),
        nullable=False,
        index=True,
    )
    to_user_id: Mapped[int] = mapped_column(
        ForeignKey("users.id", ondelete="RESTRICT"),
        nullable=False,
        index=True,
    )

    from_department: Mapped[Optional[str]] = mapped_column(String(100), nullable=True)
    to_department: Mapped[Optional[str]] = mapped_column(String(100), nullable=True)

    action: Mapped[str] = mapped_column(String(50), nullable=False, default="transfer")
    stage: Mapped[Optional[str]] = mapped_column(String(100), nullable=True)

    note: Mapped[Optional[str]] = mapped_column(Text, nullable=True)

    is_received: Mapped[bool] = mapped_column(Boolean, nullable=False, default=False)
    received_at: Mapped[Optional[datetime]] = mapped_column(DateTime(timezone=True), nullable=True)

    created_at: Mapped[datetime] = mapped_column(
        DateTime(timezone=True),
        nullable=False,
        server_default=func.now(),
    )
    updated_at: Mapped[datetime] = mapped_column(
        DateTime(timezone=True),
        nullable=False,
        server_default=func.now(),
        onupdate=func.now(),
    )

    repair_order = relationship("RepairOrder")
    from_user = relationship("User", foreign_keys=[from_user_id])
    to_user = relationship("User", foreign_keys=[to_user_id])

from __future__ import annotations

from datetime import datetime, timezone
from typing import List, Optional

from fastapi import HTTPException, status
from sqlalchemy.orm import Session

from app.models.repair_order import RepairOrder
from app.models.user import User
from app.models.workflow_transition import WorkflowTransition


class WorkflowService:
    @staticmethod
    def _get_repair_order(db: Session, repair_order_id: int) -> RepairOrder:
        repair_order = db.query(RepairOrder).filter(RepairOrder.id == repair_order_id).first()
        if not repair_order:
            raise HTTPException(
                status_code=status.HTTP_404_NOT_FOUND,
                detail="پرونده تعمیر پیدا نشد",
            )
        return repair_order

    @staticmethod
    def _get_user(db: Session, user_id: int) -> User:
        user = db.query(User).filter(User.id == user_id).first()
        if not user:
            raise HTTPException(
                status_code=status.HTTP_404_NOT_FOUND,
                detail="کاربر مقصد پیدا نشد",
            )
        return user

    @staticmethod
    def transfer_order(
        db: Session,
        repair_order_id: int,
        from_user_id: int,
        to_user_id: int,
        to_department: Optional[str] = None,
        note: Optional[str] = None,
        from_department: Optional[str] = None,
        stage: Optional[str] = None,
    ) -> WorkflowTransition:
        repair_order = WorkflowService._get_repair_order(db, repair_order_id)
        WorkflowService._get_user(db, to_user_id)

        if from_user_id == to_user_id:
            raise HTTPException(
                status_code=status.HTTP_400_BAD_REQUEST,
                detail="انتقال به خود کاربر مجاز نیست",
            )

        if repair_order is None:
            raise HTTPException(
                status_code=status.HTTP_404_NOT_FOUND,
                detail="پرونده معتبر نیست",
            )

        transition = WorkflowTransition(
            repair_order_id=repair_order_id,
            from_user_id=from_user_id,
            to_user_id=to_user_id,
            from_department=from_department,
            to_department=to_department,
            action="transfer",
            stage=stage,
            note=note,
            is_received=False,
            received_at=None,
        )

        db.add(transition)
        db.commit()
        db.refresh(transition)
        return transition

    @staticmethod
    def get_my_tasks(
        db: Session,
        current_user_id: int,
    ) -> List[WorkflowTransition]:
        return (
            db.query(WorkflowTransition)
            .filter(
                WorkflowTransition.to_user_id == current_user_id,
                WorkflowTransition.is_received.is_(False),
            )
            .order_by(WorkflowTransition.created_at.asc())
            .all()
        )

    @staticmethod
    def receive_transition(
        db: Session,
        transition_id: int,
        current_user_id: int,
    ) -> WorkflowTransition:
        transition = (
            db.query(WorkflowTransition)
            .filter(WorkflowTransition.id == transition_id)
            .first()
        )

        if not transition:
            raise HTTPException(
                status_code=status.HTTP_404_NOT_FOUND,
                detail="انتقال پیدا نشد",
            )

        if transition.to_user_id != current_user_id:
            raise HTTPException(
                status_code=status.HTTP_403_FORBIDDEN,
                detail="شما گیرنده این انتقال نیستید",
            )

        if transition.is_received:
            raise HTTPException(
                status_code=status.HTTP_400_BAD_REQUEST,
                detail="این انتقال قبلاً دریافت شده است",
            )

        transition.is_received = True
        transition.received_at = datetime.now(timezone.utc)

        db.add(transition)
        db.commit()
        db.refresh(transition)
        return transition

    @staticmethod
    def get_order_history(
        db: Session,
        repair_order_id: int,
    ) -> List[WorkflowTransition]:
        return (
            db.query(WorkflowTransition)
            .filter(WorkflowTransition.repair_order_id == repair_order_id)
            .order_by(WorkflowTransition.created_at.asc())
            .all()
        )

    @staticmethod
    def change_recipient(
        db: Session,
        transition_id: int,
        current_user_id: int,
        to_user_id: int,
        to_department: Optional[str] = None,
    ) -> WorkflowTransition:
        transition = (
            db.query(WorkflowTransition)
            .filter(WorkflowTransition.id == transition_id)
            .first()
        )

        if not transition:
            raise HTTPException(
                status_code=status.HTTP_404_NOT_FOUND,
                detail="انتقال پیدا نشد",
            )

        if transition.from_user_id != current_user_id:
            raise HTTPException(
                status_code=status.HTTP_403_FORBIDDEN,
                detail="فقط فرستنده می‌تواند گیرنده را تغییر دهد",
            )

        if transition.is_received:
            raise HTTPException(
                status_code=status.HTTP_400_BAD_REQUEST,
                detail="پس از دریافت، تغییر گیرنده مجاز نیست",
            )

        if to_user_id == transition.from_user_id:
            raise HTTPException(
                status_code=status.HTTP_400_BAD_REQUEST,
                detail="گیرنده نمی‌تواند همان فرستنده باشد",
            )

        WorkflowService._get_user(db, to_user_id)

        transition.to_user_id = to_user_id
        transition.to_department = to_department

        db.add(transition)
        db.commit()
        db.refresh(transition)
        return transition

from datetime import datetime
from fastapi import APIRouter, Depends, HTTPException, status

    WorkflowTransferRequest,

    WorkflowTaskResponse,

router = APIRouter(prefix="/api/workflow", tags=["Workflow"])


@router.post(
    "/orders/{repair_order_id}/transfer",
    response_model=WorkflowTransitionResponse,
)
def transfer_order(
    repair_order_id: int,
    payload: WorkflowTransferRequest,
    db: Session = Depends(get_db),
    current_user: User = Depends(get_current_user),
):
    order = (
        db.query(RepairOrder)
        .filter(RepairOrder.id == repair_order_id)
        .first()
    )
    if not order:
        raise HTTPException(status_code=404, detail="پرونده یافت نشد")

    if payload.to_user_id == current_user.id:
        raise HTTPException(status_code=400, detail="نمی‌توانید پرونده را به خودتان تحویل دهید")

    recipient = db.query(User).filter(User.id == payload.to_user_id).first()
    if not recipient:
        raise HTTPException(status_code=404, detail="گیرنده یافت نشد")

    # اگر فیلد department روی User دارید، این کنترل را فعال نگه دارید
    if getattr(current_user, "department", None) and getattr(recipient, "department", None):
        if payload.to_department and recipient.department != payload.to_department:
            raise HTTPException(status_code=400, detail="گیرنده انتخابی در این بخش نیست")

    # جلوگیری از چند تحویل باز هم‌زمان
    open_transition = (
        db.query(WorkflowTransition)
        .filter(
            WorkflowTransition.repair_order_id == repair_order_id,
            WorkflowTransition.status == "PENDING",
        )
        .first()
    )
    if open_transition:
        raise HTTPException(status_code=400, detail="برای این پرونده یک تحویل باز وجود دارد")

    transition = WorkflowTransition(
        repair_order_id=repair_order_id,
        from_user_id=current_user.id,
        to_user_id=payload.to_user_id,
        from_department=getattr(current_user, "department", None),
        to_department=payload.to_department,
        note=payload.note,
        status="PENDING",
        created_at=datetime.utcnow(),
    )
    db.add(transition)

    notification = Notification(
        user_id=payload.to_user_id,
        title="پرونده جدید برای دریافت",
        message=f"پرونده #{repair_order_id} برای شما ارسال شد و نیاز به تأیید دریافت دارد.",
        notification_type="WORKFLOW_TRANSFER",
        is_read=False,
        repair_order_id=repair_order_id,
        created_at=datetime.utcnow(),
    )
    db.add(notification)

    history = StatusHistory(
        repair_order_id=repair_order_id,
        old_status=getattr(order, "status", None),
        new_status=getattr(order, "status", None),
        changed_by=current_user.id,
        operator_name=getattr(current_user, "full_name", None),
        note=f"تحویل از {getattr(current_user, 'full_name', '')} به کاربر #{payload.to_user_id} در بخش {payload.to_department}",
        changed_at=datetime.utcnow(),
    )
    db.add(history)

    db.commit()
    db.refresh(transition)

    return WorkflowTransitionResponse(
        transition_id=transition.id,
        repair_order_id=transition.repair_order_id,
        from_user_id=transition.from_user_id,
        to_user_id=transition.to_user_id,
        from_department=transition.from_department,
        to_department=transition.to_department,
        action="transfer",
        note=transition.note,
        is_received=False,
        received_at=None,
        created_at=transition.created_at,
    )


@router.get("/my-tasks", response_model=list[WorkflowTaskResponse])
def my_tasks(
    db: Session = Depends(get_db),
    current_user: User = Depends(get_current_user),
):
    tasks = (
        db.query(WorkflowTransition)
        .filter(
            WorkflowTransition.to_user_id == current_user.id,
            WorkflowTransition.status == "PENDING",
        )
        .order_by(WorkflowTransition.created_at.desc())
        .all()
    )

    return [
        WorkflowTaskResponse(
            transition_id=t.id,
            repair_order_id=t.repair_order_id,
            from_user_id=t.from_user_id,
            to_user_id=t.to_user_id,
            from_department=t.from_department,
            to_department=t.to_department,
            note=t.note,
            is_received=False,
            created_at=t.created_at,
        )
        for t in tasks
    ]


@router.post("/transitions/{transition_id}/receive", response_model=WorkflowTransitionResponse)
def receive_transition(
    transition_id: int,
    db: Session = Depends(get_db),
    current_user: User = Depends(get_current_user),
):
    transition = (
        db.query(WorkflowTransition)
        .filter(WorkflowTransition.id == transition_id)
        .first()
    )
    if not transition:
        raise HTTPException(status_code=404, detail="رکورد انتقال یافت نشد")

    if transition.to_user_id != current_user.id:
        raise HTTPException(status_code=403, detail="شما گیرنده این پرونده نیستید")

    if transition.status != "PENDING":
        raise HTTPException(status_code=400, detail="این انتقال قبلاً تعیین تکلیف شده است")

    order = (
        db.query(RepairOrder)
        .filter(RepairOrder.id == transition.repair_order_id)
        .first()
    )
    if not order:
        raise HTTPException(status_code=404, detail="پرونده اصلی یافت نشد")

    transition.status = "RECEIVED"
    transition.is_received = True
    transition.received_at = datetime.utcnow()
    transition.received_by = current_user.id

    history = StatusHistory(
        repair_order_id=order.id,
        old_status=getattr(order, "status", None),
        new_status=getattr(order, "status", None),
        changed_by=current_user.id,
        operator_name=getattr(current_user, "full_name", None),
        note=f"دریافت تأیید شد از بخش {transition.from_department}",
        changed_at=datetime.utcnow(),
    )
    db.add(history)

    db.commit()
    db.refresh(transition)

    return WorkflowTransitionResponse(
        transition_id=transition.id,
        repair_order_id=transition.repair_order_id,
        from_user_id=transition.from_user_id,
        to_user_id=transition.to_user_id,
        from_department=transition.from_department,
        to_department=transition.to_department,
        action="receive",
        note=transition.note,
        is_received=True,
        received_at=transition.received_at,
        created_at=transition.created_at,
    )
